#include <Servo.h> #cargar librería para arduino

Servo cabeza; #se crea objeto cabeza con la función Servo

// Pines del sensor
int trigPin = 4; # envía pulso
int echoPin = 7; # recibe

// Pines del L298N
int IN1 = 8;   // Motor A
int IN2 = 9;
int IN3 = 10;  // Motor B
int IN4 = 11;

# recordar que digitalWrite una voltaje HIGH o LOW (5 o 0V)

// Función para medir distancia
long medirDistancia() { #Se usa long por que int solo llega a 32.767 y LONg llega a 2,147,483,647, entonces lo tiempos medidos pueden ser superios s 30000
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duracion = pulseIn(echoPin, HIGH, 30000); #echoPin (pin que recibe el sensor; medimos cuanto está en HIGH; timeout [tiempo máximo] en ms)
#PulseIn espera a que el pin pase por high y de ahí comienza a contar los ms, cuando está en low se detiene.
  long distancia = duracion * 0.034 / 2;

  if (distancia > 400) distancia = 400;  // filtro anti-ruido #El HC‑SR04 solo mide bien hasta ~400 cm

  return distancia;
}

# Usamos Void para crear nuestras propias funciones, así evitamos reescribir el código y usamos solo la función.

// Movimiento del robot
void avanzar() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void retroceder() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void detener() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void girarIzquierda() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void girarDerecha() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void setup() { #se ejectua una vez y es para emcemder la placa conm pines que serán o entrada o salida
  Serial.begin(9600);

  cabeza.attach(3);   // Servo en D3
  cabeza.write(90);   // Centro en 90 grados

  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  Serial.println("Robot explorador iniciado...");
}

void loop() { # el código que controla el funcionamiento del robot

  // Mirar al centro
  cabeza.write(90);
  delay(500);
  long centro = medirDistancia();
  Serial.print("Centro: ");
  Serial.println(centro);

  // Si hay obstáculo cerca, detener y decidir
  if (centro < 25) {
    detener();
    delay(300);

    // Mirar izquierda
    cabeza.write(20);
    delay(600);
    long izquierda = medirDistancia();
    Serial.print("Izquierda: ");
    Serial.println(izquierda);

    // Mirar derecha
    cabeza.write(160);
    delay(600);
    long derecha = medirDistancia();
    Serial.print("Derecha: ");
    Serial.println(derecha);

    // Elegir el camino más libre
    if (izquierda > derecha) {
      Serial.println("Girando a la izquierda...");
      girarIzquierda();
      delay(500);
    } else {
      Serial.println("Girando a la derecha...");
      girarDerecha();
      delay(500);
    }

    detener();
    delay(200);
  } else {
    // Avanzar si no hay obstáculo
    avanzar();
  }

  delay(100);
}
